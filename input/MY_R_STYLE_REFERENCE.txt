*******************************************************
* Figure 2-2-2. Annual Premiums and Premium Rate
* Crop Disaster Insurance
*******************************************************

clear all
set more off

* Font setting
graph set window fontface "Arial Narrow"

*******************************************************
* 1. Enter data
*******************************************************

input year premium_after_refunds premium_rate
2001     3016     3.68
2005    54847     9.25
2010    86357     5.52
2015   308477     6.14
2016   344669     4.89
2017   334252     3.94
2018   550149     4.23
2019   511146     3.26
2020   722223     3.62
2021   848524     3.80
2022   949656     3.63
2023  1039681     3.56
2024  1141145     3.62
end

*******************************************************
* 2. Create x-axis and labels
*******************************************************

gen x = _n

gen premium_label = string(premium_after_refunds, "%12.0fc")
replace premium_label = strtrim(premium_label)

gen rate_label = string(premium_rate, "%4.2f")
replace rate_label = strtrim(rate_label)

* Label positions
gen premium_label_y = premium_after_refunds + 35000
gen rate_label_y = premium_rate + 0.35

replace rate_label_y = premium_rate + 0.75 if year == 2001
replace rate_label_y = premium_rate + 0.70 if year == 2010
replace rate_label_y = premium_rate - 0.80 if year == 2018
replace rate_label_y = premium_rate + 0.20 if year == 2019
*******************************************************
* 3. Draw graph
*******************************************************

#delimit ;

twoway
    (bar premium_after_refunds x,
        yaxis(1)
        barwidth(0.55)
        fcolor("153 199 164")
        lcolor(none))

    (scatter premium_label_y x,
        yaxis(1)
        msymbol(none)
        mlabel(premium_label)
        mlabcolor(black)
        mlabsize(small)
        mlabposition(12))

    (connected premium_rate x,
        yaxis(2)
        lcolor(cranberry)
        mcolor(cranberry)
        msymbol(O)
        msize(small)
        lwidth(medthin))

    (scatter rate_label_y x,
        yaxis(2)
        msymbol(none)
        mlabel(rate_label)
        mlabcolor(cranberry)
        mlabsize(small)
        mlabposition(12)),

    title("[Figure 2-2-2] Annual Premiums and Premium Rate",
        size(medium)
        color(black))

    xlabel(1 "2001" 2 "2005" 3 "2010" 4 "2015" 5 "2016" 6 "2017"
           7 "2018" 8 "2019" 9 "2020" 10 "2021" 11 "2022"
           12 "2023" 13 "2024",
           labsize(small))

    ylabel(0(200000)1200000,
        axis(1)
        format(%12.0fc)
        labsize(small)
        angle(0)
        grid
        glcolor(gs12)
        glpattern(dash))

    ylabel(0(1)10,
        axis(2)
        format(%4.2f)
        labsize(small)
        angle(0))

    ytitle("KRW million",
        axis(1)
        size(small))

    ytitle("%",
        axis(2)
        size(small))

    xtitle("")

    legend(order(1 "Premiums after Refunds" 3 "Premium Rate (%)")
        position(12)
        ring(1)
        row(1)
        size(small)
        region(lcolor(white)))

    note("Source: NH NongHyup Property & Casualty Insurance, Agricultural Insurance Department",
        size(small)
        position(7))

    graphregion(color(white) lcolor(navy) lwidth(thin))
    plotregion(color(white) lcolor(white))
    bgcolor(white)

    xscale(range(0.4 13.6))
    yscale(axis(1) range(0 1250000))
    yscale(axis(2) range(0 10));

#delimit cr

*******************************************************
* 4. Export graph
*******************************************************

graph export "figure_2_2_2_annual_premiums_premium_rate.png", replace width(2400)
